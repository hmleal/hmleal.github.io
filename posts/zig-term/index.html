Zig Term

I would like to introduce you to zig-term a small, cross-platform terminal library for Zig, inspired by Go’s x/term. The goal is to make common terminal operations simple and approachable without requiring users to deal directly with platform-specific APIs such as termios or Windows Console APIs.

How to add it to your own project

Please use zig fetch

Examples

How to print terminal width and height

const std = @import("std");
const Io = std.Io;

const term = @import("term");

pub fn main(init: std.process.Init) !void {
    const io = init.io;

    const stdin = std.Io.File.stdin();
    const size = try term.getSize(stding);

    std.log.info("Terminal {d}x{d}", .{size.width, size.heigth});
}

How to enter the raw mode and leave pressing q or CTRL + C

const std = @import("std");
const Io = std.Io;

const term = @import("term");

pub fn main(init: std.process.Init) !void {
    const io = init.io;

    const stdin = std.Io.File.stdin();
    const size = try term.getSize(stding);
    const state = try term.makeRaw(stdin);
    defer term.restore(stdin, state) catch {};

    var running = std.atomic.Value(bool).init(true);

    // Start a new thread to check for the interruption signal
    const thread = try std.Thread.spawn(.{}, readInput, .{ io, stdin, &running });
    defer thread.join();
}

fn readInput(io: std.Io, stdin: std.Io.File, running: *std.atomic.Value(bool)) void {
    var read_buffer: [1]u8 = undefined;
    var stdin_file_reader: Io.File.Reader = .init(stdin, io, &read_buffer);

    while (running.load(.acquire)) {
        const byte = stdin_file_reader.interface.takeByte() catch return;

        if (byte == 3 or byte == 'q') {
            running.store(false, .release);
            return;
        }
    }
}
